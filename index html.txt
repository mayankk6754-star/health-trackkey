        <!-- Input Controls -->
        <div class="input-section">
            <div class="input-row">
                <input type="number" id="step-input" placeholder="Add steps">
                <button onclick="updateSteps()">+ Steps</button>
            </div>
            
            <div class="input-row">
                <input type="text" id="bp-input" placeholder="BP (e.g. 120/80)">
                <button class="btn-bp" onclick="updateBP()">Set BP</button>
            </div>

            <div class="input-row">
                <input type="number" id="water-input" placeholder="Glasses">
                <button class="btn-water" onclick="updateWater()">+ Water</button>
            </div>

            <div class="input-row">
                <input type="number" id="active-input" placeholder="Mins">
                <button class="btn-active" onclick="updateActive()">+ Active</button>
            </div>

            <button class="reset-btn" onclick="resetDay()">Reset Daily Data</button>
        </div>

    <script>
        // Load saved data or set defaults
        let healthData = JSON.parse(localStorage.getItem('healthData')) || {
            steps: 0,
            bp: '--/--',
            water: 0,
            activeMinutes: 0
        };

        // Display data on load
        function renderData() {
            document.getElementById('display-steps').innerText = `${healthData.steps.toLocaleString()} / 10,000`;
            document.getElementById('display-bp').innerText = `${healthData.bp} mmHg`;
            document.getElementById('display-water').innerText = `${healthData.water} Glasses`;
            document.getElementById('display-active').innerText = `${healthData.activeMinutes} Mins`;
        }

        function saveData() {
            localStorage.setItem('healthData', JSON.stringify(healthData));
            renderData();
        }

        function updateSteps() {
            const val = parseInt(document.getElementById('step-input').value);
            if (!isNaN(val)) {
                healthData.steps += val;
                document.getElementById('step-input').value = '';
                saveData();
            }
        }

        function updateBP() {
            const val = document.getElementById('bp-input').value.trim();
            if (val !== "") {
                healthData.bp = val;
                document.getElementById('bp-input').value = '';
                saveData();
            }
        }

        function updateWater() {
            const val = parseInt(document.getElementById('water-input').value);
            if (!isNaN(val)) {
                healthData.water += val;
                document.getElementById('water-input').value = '';
                saveData();
            }
        }

        function updateActive() {
            const val = parseInt(document.getElementById('active-input').value);
            if (!isNaN(val)) {
                healthData.activeMinutes += val;
                document.getElementById('active-input').value = '';
                saveData();
            }
        }

        function resetDay() {
            if (confirm("Are you sure you want to reset your daily tracking data?")) {
                healthData = { steps: 0, bp: '--/--', water: 0, activeMinutes: 0 };
                saveData();
            }
        }

        // Run on initial load
        renderData();
    </script>
